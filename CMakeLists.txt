cmake_minimum_required(VERSION 3.16)
project(BuildingEdgeBMSLab VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BMS_SANITIZE "Enable AddressSanitizer and UBSan (Linux clang/gcc)" OFF)
add_library(bms_core STATIC cpp/src/core.cpp cpp/src/net.cpp)
target_include_directories(bms_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cpp/include)
target_compile_options(bms_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(BMS_SANITIZE)
  target_compile_options(bms_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(bms_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(bms-edge cpp/src/main.cpp)
target_link_libraries(bms-edge PRIVATE bms_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(bms-core-tests cpp/tests/test_core.cpp)
  target_link_libraries(bms-core-tests PRIVATE bms_core)
  target_compile_options(bms-core-tests PRIVATE -UNDEBUG)
  add_test(NAME bms-core-tests COMMAND bms-core-tests)
endif()
